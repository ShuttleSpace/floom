**Tech Stack:** Rust（`floom-tsx`、`floom-dsl`、`floom-cli`），swc_ecma_parser（已有），TypeScript（仅用于 smoke 校验，非依赖）。

  ```rust
  // SPDX-License-Identifier: MIT
  //! 生成 `.d.ts` 编辑器声明，覆盖整个解析器接受的 API 面。
  //!
  //! 与 `floom-dsl` 的 schema 表同源：每个 `NodeKind` 一个 interface，props
  //! 与 events 分别来自 `prop_spec` 与 `events_for`，带 domain 的属性降级为
  //! 字面量联合。生成文件是纯声明（无顶层 import/export），因此
  //! `declare module "floom"` 是真正的环境模块声明，`declare namespace JSX`
  //! 全局生效。

  use floom_dsl::ui_ir::NativeKind;
  use floom_dsl::ui_ir::{
      EventName, NodeKind, PropSpec, ValueType, all_prop_names, events_for, prop_spec,
  };

  /// 生成一份完整的 `.d.ts` 字符串（全量，不依赖某个文档）。
  pub fn generate() -> String {
      let mut out = String::from("// @generated by floem-tsx. Do not edit.\n\n");

      out.push_str(&floom_namespace_open());
      for kind in NodeKind::ALL {
          out.push_str(&interface_for(kind));
          out.push('\n');
      }
      out.push_str(&app_props_interface());
      out.push_str("}\n\n");

      out.push_str(&jsx_namespace_open());
      for kind in NodeKind::ALL {
          out.push_str(&format!("    {}: Floom.{}Props;\n", tag_name(kind), kind_ts_name(kind)));
      }
      out.push_str("    App: Floom.AppProps;\n");
      out.push_str("  }\n}\n\n");

      out.push_str(FLOOM_MODULE);
      out
  }

  /// `declare namespace Floom {` 开头 + 共享类型 + 每个组件 interface 之前的部分。
  fn floom_namespace_open() -> String {
      String::from(
          "declare namespace Floom {\n  \
           export interface Accessor<T> { (): T }\n  \
           export interface Setter<T> { (value: T): void }\n  \
           export type Component = () => JSX.Element;\n  \
           // 窄于解析器（解析器也接受无 # 前缀）：用 # 模板字面量挡掉 \
           background=\"red\" 这类解析器会拒绝的字面量。\n  \
           export type Color = `#${string}`;\n",
      )
  }

  /// 单个组件的 props 接口。
  fn interface_for(kind: NodeKind) -> String {
      let mut lines = vec![format!("  export interface {}Props {{", kind_ts_name(kind))];
      for name in all_prop_names() {
          if let Some(spec) = prop_spec(kind, name) {
              lines.push(format!("    {}?: {};", name, ts_type(&spec)));
          }
      }
      lines.push("    class?: string;".to_string());
      if kind.is_container() || kind == NodeKind::Text {
          lines.push("    children?: JSX.Element | JSX.Element[];".to_string());
      }
      if is_placeholder(kind) {
          lines.push(
              "    // 当前 emitter 以占位方式挂载此组件：props/events 可解析，但可能未接线。"
                  .to_string(),
          );
      }
      for event in events_for(kind) {
          lines.push(format!(
              "    {}?: {};",
              event_handler_name(event),
              event_handler_type(kind, event)
          ));
      }
      lines.push("  }".to_string());
      lines.join("\n")
  }

  /// `<App title size width height>` 包装根，解析器在 `entry::config_from_wrapper`
  /// 接受这些属性，但它不是 `NodeKind`，故单列。
  fn app_props_interface() -> String {
      String::from(
          "  export interface AppProps {\n    \
           title?: string;\n    size?: string;\n    width?: string;\n    \
           height?: string;\n    children?: JSX.Element;\n  }",
      )
  }

  fn jsx_namespace_open() -> String {
      String::from(
          "declare namespace JSX {\n  \
           interface Element {}\n  \
           interface ElementChildrenAttribute { children: {} }\n  \
           interface IntrinsicElements {\n",
      )
  }

  /// TSX 标签名（也是 `IntrinsicElements` 的键）：原生组件用解析器接受的拼写
  /// （`Web`/`Video`/`Map`/`TextInput`），其余用 `name()` 的 PascalCase。
  fn tag_name(kind: NodeKind) -> String {
      kind_ts_name(kind)
  }

  fn kind_ts_name(kind: NodeKind) -> String {
      match kind {
          NodeKind::Native(NativeKind::Web) => "Web".to_string(),
          NodeKind::Native(NativeKind::Video) => "Video".to_string(),
          NodeKind::Native(NativeKind::Map) => "Map".to_string(),
          NodeKind::Native(NativeKind::TextInput) => "TextInput".to_string(),
          other => other
              .name()
              .split('_')
              .map(|s| {
                  let mut c = s.chars();
                  match c.next() {
                      Some(first) => first.to_uppercase().collect::<String>() + c.as_str(),
                      None => String::new(),
                  }
              })
              .collect(),
      }
  }

  /// 属性值类型；带 domain 的 `Ident` 降级为字面量联合。
  fn ts_type(spec: &PropSpec) -> String {
      match (spec.ty, spec.domain.is_empty()) {
          (ValueType::Color, _) => "Floom.Color".to_string(),
          (ValueType::Bool, _) => "boolean".to_string(),
          (ValueType::Int, _) | (ValueType::Number, _) => "number".to_string(),
          (ValueType::Text, _) => "string".to_string(),
          (ValueType::Ident, false) => spec
              .domain
              .iter()
              .map(|s| format!("\"{s}\"", s = s))
              .collect::<Vec<_>>()
              .join(" | "),
          (ValueType::Ident, true) => "string".to_string(),
          (ValueType::List, _) => "unknown[]".to_string(),
      }
  }

  /// 事件处理器签名，按真正消费它的 builder 来定。
  fn event_handler_type(kind: NodeKind, event: EventName) -> String {
      match (kind, event) {
          (NodeKind::Radio, EventName::Toggle) => "() => void".to_string(),
          (NodeKind::Checkbox | NodeKind::Switch, EventName::Toggle) => {
              "(checked: boolean) => void".to_string()
          }
          (NodeKind::Slider, EventName::Change) => "(value: number) => void".to_string(),
          (NodeKind::Select | NodeKind::Tabs, EventName::Change) => {
              "(index: number) => void".to_string()
          }
          (NodeKind::TextField, EventName::Input | EventName::Submit) => {
              "(value: string) => void".to_string()
          }
          (NodeKind::List, EventName::Scroll) => "(offset: number) => void".to_string(),
          _ => "() => void".to_string(),
      }
  }

  /// `Click` → `onClick`（camelCase）。
  fn event_handler_name(event: EventName) -> String {
      let name = event.name();
      let mut chars = name.chars();
      match chars.next() {
          Some(first) => format!("on{}{}", first.to_uppercase(), chars.as_str()),
          None => "on".to_string(),
      }
  }

  /// emitter 以占位方式挂载的组件：props/events 可解析但未接线。
  fn is_placeholder(kind: NodeKind) -> bool {
      matches!(
          kind,
          NodeKind::Grid
              | NodeKind::List
              | NodeKind::Overlay
              | NodeKind::Dialog
              | NodeKind::Popup
              | NodeKind::Tabs
              | NodeKind::NavigationBar
              | NodeKind::NavigationRail
              | NodeKind::NavigationDrawer
              | NodeKind::Image
      )
  }

  const FLOOM_MODULE: &str = r#"
  declare module "floom" {
    export function createSignal<T>(value: T): [Floom.Accessor<T>, Floom.Setter<T>];
    export function createMemo<T>(fn: () => T): Floom.Accessor<T>;
    export function createEffect(fn: () => void): void;
    export function provideContext<T>(name: string, value: T): void;
    export function useContext<T = string>(name: string): T;
    export type Component = Floom.Component;
    export function mount(component: Component): void;
    export const app: { setName(name: string): void };
    export const window: {
      setTitle(title: string): void;
      setSize(width: number, height: number): void;
      setSize(size: string): void;
      setResizable(resizable: boolean): void;
    };
    export const platform: {
      /** @param capability windowing | clipboard | ime | accessibility | webgpu | software_render | touch | native_view */
      require(capability: Capability, onUnsupported: Policy): void;
    };
    export type Capability = "windowing" | "clipboard" | "ime" | "accessibility" | "webgpu" | "software_render" | "touch" | "native_view";
    export type Policy = "warn" | "fail" | "exit";
  }
  "#;
  ```

  ```rust
  /// 抽取 `IntrinsicElements { ... }` 里的 `Key: Floom.XxxProps;` 的 Key。
  fn intrinsic_tags(dts: &str) -> Vec<String> {
      let mut out = Vec::new();
      let mut in_block = false;
      for line in dts.lines() {
          if line.trim_start().starts_with("interface IntrinsicElements") {
              in_block = true;
              continue;
          }
          if in_block {
              let t = line.trim();
              if t == "}" { break; }
              if let Some((key, _)) = t.split_once(':') {
                  out.push(key.trim().to_string());
              }
          }
      }
      out
  }
  /// 抽取某个 `interface XxxProps { ... }` 内的 `name?: type;` 字段名。
  fn field_names(dts: &str, interface: &str) -> Vec<String> {
      let marker = format!("interface {interface}");
      let mut out = Vec::new();
      let mut depth: i32 = 0;
      let mut in_iface = false;
      for line in dts.lines() {
          let t = line.trim_start();
          if t.starts_with(&marker) {
              in_iface = true;
              depth = 0;
              continue;
          }
          if in_iface {
              depth += t.matches('{').count() as i32 - t.matches('}').count() as i32;
              if let Some((name, _)) = t.split_once(':') {
                  let name = name.trim().trim_end_matches('?');
                  if !name.is_empty()
                      && name.chars().next().map(|c| c.is_alphabetic()).unwrap_or(false)
                  {
                      out.push(name.to_string());
                  }
              }
              if depth <= 0 { break; }
          }
      }
      out
  }
  /// 抽取所有 `interface XxxProps` 的名字。
  fn interface_names(dts: &str) -> Vec<String> {
      let mut out = Vec::new();
      for line in dts.lines() {
          let t = line.trim_start();
          if let Some(rest) = t.strip_prefix("interface ") {
              if let Some(name) = rest.split_whitespace().next() {
                  if name.ends_with("Props") {
                      out.push(name.to_string());
                  }
              }
          }
      }
      out
  }

  fn ts_name(kind: floem_dsl::ui_ir::NodeKind) -> String {
      use floem_dsl::ui_ir::NativeKind;
      use floem_dsl::ui_ir::NodeKind;
      match kind {

              match c.next() {

                  None => String::new(),

          }).collect(),
      }
  }
  ```

  ```rust
      // 这些 TS 类型别名由 .d.ts 暴露给编辑器；解析器是 ES-only（不接受 TS 注解），
      // 故只验证声明文本包含它们，不通过 parser 编译验证。
      let dts = floem_tsx::generate_dts();
      assert!(dts.contains("export type Component = () => JSX.Element"));
      assert!(dts.contains("export interface Accessor<T>"));
      assert!(dts.contains("export interface Setter<T>"));
      assert!(dts.contains("export type Color = `#${string}`"));
  }

  #[test] fn parity_full_document_roundtrip() {
      // 用从生成文本里抽出的每个 tag 拼出一段 TSX，整体编译必须成功。
      // 每个组件用最简 <Tag />（schema 属性都是可选的），容器无子节点也可解析；
      // 属性/事件的逐个接受已由 parity_prop_accepted / parity_event_accepted 覆盖。
      let dts = floem_tsx::generate_dts();
      let mut body = String::new();
      for tag in intrinsic_tags(&dts) {
          if tag == "App" { continue; }
          body += &format!("<{tag} />");
      }
      let src = format!(
          "import {{ mount }} from \"floom\";\nfunction App() {{ return <Stack>{body}</Stack>; }}\nmount(App);"
      );
      floem_tsx::compile(&src).expect("every declared component must be parseable");
  }

  #[test] fn parity_encouraged_syntax() {
      // 解析器是 ES+JSX（不接受 TS 类型注解），因此只验证 .d.ts 鼓励的*运行时*用法
      // 能被 parser 接受：解构 createSignal、函数组件、mount。TS 注解形式的覆盖见
      // decl_exposes_reactive_type_aliases（声明文本存在性）。
      let src = r#"
        import { createSignal, mount } from "floom";
        const [n, setN] = createSignal(0);
        function App() { return <Stack />; }
        mount(App);
      "#;
      floem_tsx::compile(src).expect("encouraged runtime usage must parse");
  }
  ```
  > `parity_full_document_roundtrip` 用最简 `<Tag />` 拼装，避免给每个属性塞样本值导致的类型/子树噪声；属性与事件的逐个接受分别由 `parity_prop_accepted` / `parity_event_accepted` 证明，整体仍满足「声明 ⊆ 解析器接受」。
